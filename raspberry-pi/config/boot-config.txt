# /boot/firmware/config.txt 맨 아래 [all] 섹션에 이 세 줄이 있어야 한다.
# 줄 앞이나 '=' 양옆에 공백을 넣지 말 것.
#
# enable_uart=1        PMS5003용 UART 활성화
# dtoverlay=disable-bt 파이4 기본 시리얼(mini-UART)은 속도가 흔들려 PMS5003 데이터가 깨진다.
#                      블루투스가 쓰던 안정적인 UART(PL011)를 센서에 넘긴다.
# dtoverlay=i2c-gpio   VL53L0X 2대가 같은 주소(0x29)라 GPIO23(SDA)/GPIO24(SCL)로
#                      I2C 버스 3번을 하나 더 만든다.

[all]
enable_uart=1
dtoverlay=disable-bt
dtoverlay=i2c-gpio,bus=3,i2c_gpio_sda=23,i2c_gpio_scl=24
